cmake_minimum_required(VERSION 3.22)
project(LocalSplitScreenTools LANGUAGES C)
file(GLOB LUA_CORE CONFIGURE_DEPENDS "vendor/lua-5.4.8/src/*.c")
list(FILTER LUA_CORE EXCLUDE REGEX "/lua(c)?\\.c$")
add_library(lua_core STATIC ${LUA_CORE})
target_include_directories(lua_core PUBLIC vendor/lua-5.4.8/src)
add_executable(lua_runner vendor/lua-5.4.8/src/lua.c)
target_link_libraries(lua_runner PRIVATE lua_core)
add_library(lss_native SHARED src/native.c)
set_target_properties(lss_native PROPERTIES OUTPUT_NAME lss_native_v2)
target_include_directories(lss_native PRIVATE vendor/sdl3/SDL3-3.4.20/include)
target_link_libraries(lss_native PRIVATE lua_core "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sdl3/SDL3-3.4.20/lib/x64/SDL3.lib" delayimp)
target_link_options(lss_native PRIVATE /DELAYLOAD:SDL3.dll)
add_custom_command(TARGET lss_native POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sdl3/SDL3-3.4.20/lib/x64/SDL3.dll" "$<TARGET_FILE_DIR:lss_native>/SDL3.dll")
target_compile_definitions(lss_native PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_test(NAME persistence_and_session COMMAND lua_runner tests/run.lua)
set_tests_properties(persistence_and_session PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
