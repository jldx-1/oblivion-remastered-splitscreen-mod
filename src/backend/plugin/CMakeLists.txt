cmake_minimum_required(VERSION 3.21)
project(LocalSplitScreenBackend LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The OBSE64 plugin requires 64-bit Windows")
endif()
add_library(LocalSplitScreenBackend SHARED Backend.cpp)
target_compile_features(LocalSplitScreenBackend PRIVATE cxx_std_17)
target_compile_definitions(LocalSplitScreenBackend PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    set_property(TARGET LocalSplitScreenBackend PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(LocalSplitScreenBackend PRIVATE /W4 /permissive- /utf-8)
endif()
